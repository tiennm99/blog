A run of exactly one page shows that page instead of an ellipsis, so the ellipsis always stands for two or more pages. */ -}} {{- $p := .Paginator -}} {{- if gt $p.TotalPages 1 -}} {{- $cur := $p.PageNumber -}} {{- $last := $p.TotalPages -}} {{- $win := 2 -}} {{- $lo := sub $cur $win -}} {{- $hi := add $cur $win -}}
{{- if $p.HasPrev }} ‹ {{ i18n "prev" | default "Trước" }} {{- else }} ‹ {{ i18n "prev" | default "Trước" }} {{- end }}
    {{- $gap := false }} {{- range $p.Pagers }} {{- $n := .PageNumber }} {{- $show := or (eq $n 1) (eq $n $last) (and (ge $n $lo) (le $n $hi)) (and (eq $n 2) (eq $lo 3)) (and (eq $n (sub $last 1)) (eq $hi (sub $last 2))) }} {{- if $show }} {{- if $gap }}
  1. …
    2. {{- $gap = false }} {{- end }} {{- if eq $n $cur }}
  2. {{ $n }}
    3. {{- else }}
  3. {{ $n }}
    4. {{- end }} {{- else }} {{- $gap = true }} {{- end }} {{- end }}

{{- end -}}
