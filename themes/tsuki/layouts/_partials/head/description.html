{{- /* Returns the page's meta description as plain text, whitespace collapsed and truncated to about 160 characters. Shared by , OpenGraph, Twitter Cards and JSON-LD so all of them say the same thing. */ -}} {{- $raw := .Description | default .Summary | default site.Params.description | default site.Title -}} {{- $desc := $raw | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace | truncate 160 -}} {{- return $desc -}}
