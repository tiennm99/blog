{{- /* Prev/next post navigation within the same section. Gated by params.prevNextNav.enable (default true). Opt-in params.prevNextNav.taxonomy (for example "categories"): when the post has a term in that taxonomy, prev/next stays within the post's first term and section, so a series links to its own neighbours. Posts without a term fall back to the whole section. Silently skipped when neither a prev nor a next exists. */ -}} {{- $cfg := site.Params.prevNextNav | default dict -}} {{- if ne ($cfg.enable | default true) false -}} {{- $prev := .PrevInSection -}} {{- $next := .NextInSection -}} {{- with $cfg.taxonomy -}} {{- range first 1 ($.GetTerms .) -}} {{- $pages := where .Pages "Section" $.Section -}} {{- /* Term pages sort newest first, so Pages.Prev is the older post, like PrevInSection. */ -}} {{- $prev = $pages.Prev $ -}} {{- $next = $pages.Next $ -}} {{- end -}} {{- end -}} {{- if or $prev $next -}}
{{- if $prev }} ← {{ i18n "prevPost" | default "Previous post" }} {{ $prev.LinkTitle | default $prev.Title }} {{- else }} {{- end }} {{- if $next }} {{ i18n "nextPost" | default "Next post" }} → {{ $next.LinkTitle | default $next.Title }} {{- end }}
{{- end -}} {{- end -}}
